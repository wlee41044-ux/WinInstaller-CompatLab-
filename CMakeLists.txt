cmake_minimum_required(VERSION 3.16)
project(WinInstallerCompatibilityLab LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(WINCLAB_SANITIZERS "Build host tests with AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
if(WINCLAB_SANITIZERS)
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    message(FATAL_ERROR "WINCLAB_SANITIZERS requires GCC or Clang")
  endif()
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -fno-sanitize-recover=all)
  add_link_options(-fsanitize=address,undefined -fno-sanitize-recover=all)
endif()
add_library(winclab_file_api STATIC src/file_api.cpp)
target_include_directories(winclab_file_api PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_options(winclab_file_api PRIVATE -Wall -Wextra -Wpedantic -Werror)
include(CTest)
if(BUILD_TESTING)
  add_executable(winclab_tests tests/test_file_api.cpp)
  target_link_libraries(winclab_tests PRIVATE winclab_file_api)
  target_compile_options(winclab_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
  add_test(NAME winclab_file_compatibility COMMAND winclab_tests)
endif()

# Registry is independently linkable and has no filesystem or guest ABI dependency.
add_library(winclab_registry_api STATIC src/registry_api.cpp)
target_include_directories(winclab_registry_api PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_options(winclab_registry_api PRIVATE -Wall -Wextra -Wpedantic -Werror)
if(BUILD_TESTING)
  add_executable(winclab_file_edges tests/test_file_edges.cpp)
  target_link_libraries(winclab_file_edges PRIVATE winclab_file_api)
  target_compile_options(winclab_file_edges PRIVATE -Wall -Wextra -Wpedantic -Werror)
  foreach(case open_always_readonly invalid_crt_mode readonly_open readonly_replace cross_drive negative_seek dispositions sharing_preservation handle_paths creation_last_error flush_access)
    add_test(NAME file_${case} COMMAND winclab_file_edges ${case})
  endforeach()
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|amd64|AMD64)$")
    add_executable(winclab_registry_output_probe tests/test_registry_output_probe.cpp)
    target_link_libraries(winclab_registry_output_probe PRIVATE winclab_registry_api)
    target_compile_options(winclab_registry_output_probe PRIVATE -Wall -Wextra -Wpedantic -Werror)
    add_test(NAME registry_output_only_memory COMMAND winclab_registry_output_probe)
  endif()
  add_executable(winclab_registry_tests tests/test_registry_api.cpp)
  target_link_libraries(winclab_registry_tests PRIVATE winclab_registry_api)
  target_compile_options(winclab_registry_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
  foreach(case lifecycle data_types buffer_sizes raw_strings_default access hive_policy invalid_arguments options unicode isolation handle_lifetime buffer_ownership)
    add_test(NAME registry_${case} COMMAND winclab_registry_tests ${case})
  endforeach()
  add_executable(winclab_profiles tests/test_profiles.cpp)
  target_link_libraries(winclab_profiles PRIVATE winclab_file_api winclab_registry_api)
  target_compile_options(winclab_profiles PRIVATE -Wall -Wextra -Wpedantic -Werror)
  foreach(profile utility installer registry crt)
    add_test(NAME profile_${profile} COMMAND winclab_profiles ${profile})
  endforeach()
endif()

# Independently testable Registry delete/enumeration host operations.
if(BUILD_TESTING)
  add_executable(winclab_registry_extensions tests/test_registry_extensions.cpp)
  target_link_libraries(winclab_registry_extensions PRIVATE winclab_registry_api)
  target_compile_options(winclab_registry_extensions PRIVATE -Wall -Wextra -Wpedantic -Werror)
  add_test(NAME registry_enumerate_delete COMMAND winclab_registry_extensions)
endif()
